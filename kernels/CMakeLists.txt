cmake_minimum_required(VERSION 3.24)
project(cmk_prototypes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cmk_host_check host_check.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cmk_host_check PRIVATE -O2 -fno-fast-math -ffp-contract=off)
endif()
option(CMK_ENABLE_CUDA "Build unvalidated Hopper CUDA prototypes" OFF)
if(CMK_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 90)
  endif()
  enable_language(CUDA)
  add_executable(cmk_cuda_smoke smoke.cu)
  set_target_properties(cmk_cuda_smoke PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cmk_cuda_smoke PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
endif()
add_executable(cmk_imported_check imported_check.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cmk_imported_check PRIVATE -O2 -fno-fast-math -ffp-contract=off)
endif()
